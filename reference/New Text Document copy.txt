function getModeScore(history, bookId, mode) {
  const entries = history.filter((h) => h.bookId === bookId && h.mode === mode);
  if (entries.length === 0) return null;
  return entries[entries.length - 1].score;
}

                  {score === null ? "—" : `${score}%`}

              );

        </div>
      ))}
    </div>
  );

function computeBadgeTally(history) {
  const tally = { Bronze: 0, Silver: 0, Gold: 0 };

function StarField({ count = 40 }) {
  const stars = Array.from({ length: count }, (_, i) => {
    const seed = (i * 97) % 100;
    return {
      top: `${(i * 37) % 100}%`,
      left: `${(i * 61 + seed) % 100}%`,
      size: 1 + (i % 3),
      opacity: 0.3 + ((i * 13) % 50) / 100,
    };
  });
  return (
    <div className="absolute inset-0 overflow-hidden pointer-events-none">
      {stars.map((s, i) => (
        <div key={i} className="bk-star-dot" style={{ top: s.top, left: s.left, width: s.size, height: s.size, opacity: s.opacity }} />
      ))}
    </div>
  );
}

}

function AssessmentScreen({ book, lang, qIndex, setQIndex, answers, setAnswers, onSubmit }) {

  const answer = answers[q.id];

  return (
    <div className="max-w-xl mx-auto bg-white rounded-2xl p-8 bk-card">
      <div className="flex items-center justify-between mb-1">
        <span className="bk-mono text-[10px] bk-gold">{t(UI.questionLabel, lang)} {qIndex + 1} / {QUESTIONS.length}</span>
        <span className="bk-mono text-[10px] bk-slate">{t(UI.bloomLabel, lang)}: {q.bloom.toUpperCase()}</span>
      </div>
      <div className="w-full h-1 bg-[#EDE6D3] rounded-full mb-6 overflow-hidden">
        <div className="h-full bk-bg-gold" style={{ width: `${((qIndex + 1) / QUESTIONS.length) * 100}%` }} />
      </div>

      <h3 className="bk-display text-xl mb-6 leading-snug">{t(q.prompt, lang)}</h3>

      {q.type === "mc" && (
        <div className="flex flex-col gap-2">
          {t(q.options, lang).map((opt, i) => (
            <button
              key={i} onClick={() => setAnswer(i)}
              className={`text-left px-4 py-2.5 rounded-lg border text-sm ${answer === i ? "bk-bg-night bk-cream-text border-transparent" : "bk-border hover:bg-[#FBF7EF]"}`}
            >
              {opt}
            </button>
          ))}
        </div>
      )}

      {q.type === "ordering" && (
        <OrderingInput items={t(q.items, lang)} value={answer || q.items.en.map((_, i) => i)} onChange={setAnswer} />
      )}

      {q.type === "matching" && (
        <MatchingInput left={t(q.left, lang)} right={t(q.right, lang)} value={answer || {}} onChange={setAnswer} lang={lang} />
      )}

      {q.type === "fillblank" && (
        <input
          value={answer || ""} onChange={(e) => setAnswer(e.target.value)}
          placeholder={t(UI.typeAnswer, lang)} className="w-full border bk-border rounded-lg px-3 py-2 outline-none text-sm"
        />
      )}

            {t(UI.next, lang)} <ArrowRight size={14} />

    </div>
  );
}

function OrderingInput({ items, value, onChange }) {
  const move = (idx, dir) => {
    const next = [...value];
    const target = idx + dir;
    if (target < 0 || target >= next.length) return;
    [next[idx], next[target]] = [next[target], next[idx]];
    onChange(next);
  };
  return (
    <div className="flex flex-col gap-2">
      {value.map((itemIdx, pos) => (
        <div key={itemIdx} className="flex items-center gap-2 border bk-border rounded-lg px-3 py-2 text-sm">
          <span className="bk-mono text-[10px] bk-slate w-4">{pos + 1}</span>
          <span className="flex-1">{items[itemIdx]}</span>
          <button onClick={() => move(pos, -1)} className="bk-slate hover:bk-ink px-1">↑</button>
          <button onClick={() => move(pos, 1)} className="bk-slate hover:bk-ink px-1">↓</button>
        </div>
      ))}
    </div>
  );
}

function MatchingInput({ left, right, value, onChange, lang }) {
  return (
    <div className="flex flex-col gap-2">
      {left.map((l, i) => (
        <div key={i} className="flex items-center gap-3">
          <span className="text-sm w-40 shrink-0">{l}</span>
          <select
            value={value[i] ?? ""}
            onChange={(e) => onChange({ ...value, [i]: Number(e.target.value) })}
            className="flex-1 border bk-border rounded-lg px-2 py-1.5 text-sm outline-none"
          >
            <option value="" disabled>{t(UI.chooseMatch, lang)}</option>
            {right.map((r, j) => (
              <option key={j} value={j}>{r}</option>
            ))}
          </select>
        </div>
      ))}
    </div>
  );
}

function ClassificationInput({ q, lang, value, onChange }) {
  return (
    <div className="flex flex-col gap-2">
      {q.options.map((opt, i) => (
        <div key={i} className="flex items-center justify-between border bk-border rounded-lg px-3 py-2">

          <div className="flex gap-1.5">
            {["A", "B"].map((g) => (

/* ---------- grading ---------- */
function gradeQuestion(q, answer, lang) {
  if (answer === undefined) return 0;
  if (q.type === "mc") return answer === q.correct ? 1 : 0;
  if (q.type === "fillblank") {
    const norm = (s) => (s || "").trim().toLowerCase();
    const correctAnswer = norm(t(q.answer, lang));
    const alts = (q.altAnswers && q.altAnswers[lang]) || [];
    return norm(answer) === correctAnswer || alts.some((a) => norm(answer) === norm(a)) ? 1 : 0;
  }
  if (q.type === "ordering") {
    const correctCount = answer.filter((v, i) => v === q.correctOrder[i]).length;
    return correctCount / q.correctOrder.length;
  }
  if (q.type === "matching") {
    const total = q.left.en.length;
    let correct = 0;
    for (let i = 0; i < total; i++) if (answer[i] === q.correctMap[i]) correct++;
    return correct / total;
  }
  if (q.type === "classification") {
    const total = q.options.length;
    let correct = 0;
    q.options.forEach((opt, i) => { if (answer[i] === opt.group) correct++; });
    return correct / total;
  }
  if (q.type === "open") {
    const text = (answer || "").toLowerCase();
    const keywords = (q.keywords && q.keywords[lang]) || [];
    return keywords.some((k) => text.includes(k.toLowerCase())) ? 1 : 0;
  }
  return 0;
}

function ResultsScreen({ book, lang, answers, selectedMode, onDone }) {
  const QUESTIONS = book.questions;
  const scores = QUESTIONS.map((q) => gradeQuestion(q, answers[q.id], lang));
  const overall = Math.round((scores.reduce((a, b) => a + b, 0) / QUESTIONS.length) * 100);
  const mode = READING_MODES.find((m) => m.id === selectedMode);

  return (
    <div className="max-w-xl mx-auto">
      <div className="bk-bg-night rounded-2xl p-8 text-center bk-card mb-6 relative overflow-hidden">
        <StarField count={25} />
        <div className="relative">
          <div className={`w-14 h-14 rounded-full mx-auto flex items-center justify-center mb-4 ${mode.badgeClass}`}>
            <Trophy size={22} className="text-white" />
          </div>
          <div className="bk-mono text-[10px] bk-gold mb-2">{mode.badge.toUpperCase()} {t(UI.badgeEarned, lang)}</div>
          <h2 className="bk-display bk-cream-text text-3xl mb-1">{overall}%</h2>
          <p className="bk-cream-slate text-sm">{t(UI.overallScore, lang)} · +{mode.points} {t(UI.xpEarned, lang)}</p>
        </div>
      </div>

      <div className="flex flex-col gap-3">
        {QUESTIONS.map((q, i) => {
          const s = scores[i];
          const correct = s === 1;
          return (
            <div key={q.id} className="bg-white rounded-xl p-4 bk-card">
              <div className="flex items-start gap-3">
                <div className={`w-6 h-6 rounded-full flex items-center justify-center shrink-0 mt-0.5 ${correct ? "bk-bg-sage" : s > 0 ? "bk-bg-gold" : "bg-[#E7B0B7]"}`}>
                  {correct ? <Check size={13} className="text-white" /> : <X size={13} className="text-white" />}
                </div>
                <div className="flex-1">
                  <div className="text-sm font-medium mb-1">{t(q.prompt, lang)}</div>
                  <div className="bk-mono text-[10px] bk-slate mb-1">
                    {s === 1 ? t(UI.correct, lang) : s > 0 ? `${t(UI.partialCredit, lang)} (${Math.round(s * 100)}%)` : t(UI.notQuite, lang)} · {q.bloom}
                  </div>
                  <div className="bk-slate text-xs">{t(q.explanation, lang)}</div>
                </div>
              </div>
            </div>
          );
        })}
      </div>

      <button onClick={onDone} className="w-full bk-btn-dark rounded-lg py-2.5 text-sm font-medium mt-6">
        {t(UI.backToLibrary, lang)}
      </button>
    </div>
  );
}

function BookDetailScreen({ child, bookId, lang, onBack }) {
  const bookTitle = bookTitleById(bookId, lang);
  const modes = ["short", "extended", "physical"];
  const chartData = modes.map((m) => {
    const attemptsForMode = child.history.filter((h) => h.bookId === bookId && h.mode === m);
    const avg = attemptsForMode.length
      ? Math.round(attemptsForMode.reduce((a, h) => a + h.score, 0) / attemptsForMode.length)
      : 0;
    return { mode: t(UI[MODE_LABELS[m]], lang), score: avg, attempted: attemptsForMode.length > 0 };
  });
  const bookHistory = child.history.filter((h) => h.bookId === bookId);

  return (
    <div className="bk-bg-cream min-h-[560px] px-6 py-10">
      <div className="max-w-2xl mx-auto">
        <button onClick={onBack} className="flex items-center gap-1 bk-slate text-sm mb-6 hover:bk-ink">
          <ArrowLeft size={14} /> {t(UI.bookDetailBack, lang)}
        </button>

        <h2 className="bk-display text-2xl mb-1">{bookTitle}</h2>
        <p className="bk-slate text-sm mb-6">{child.name}</p>

        <div className="bg-white rounded-2xl p-6 bk-card mb-6">
          <div className="flex items-center gap-2 mb-4">
            <BarChart3 size={16} className="bk-gold" />
            <div className="bk-display text-lg">{t(UI.byModeHeading, lang)}</div>
          </div>
          <div style={{ width: "100%", height: 220 }}>
            <ResponsiveContainer>
              <BarChart data={chartData}>
                <CartesianGrid strokeDasharray="3 3" stroke="#EDE6D3" />
                <XAxis dataKey="mode" tick={{ fontSize: 11, fill: "#5B6B7C" }} />
                <YAxis domain={[0, 100]} tick={{ fontSize: 10, fill: "#5B6B7C" }} />
                <Tooltip />
                <Bar dataKey="score" radius={[6, 6, 0, 0]} fill="#E3A94F" />
              </BarChart>
            </ResponsiveContainer>
          </div>
          <div className="grid grid-cols-3 gap-2 mt-4">
            {chartData.map((d, i) => (
              <div key={i} className="text-center">
                <div className="bk-mono text-[9px] bk-slate">{d.mode.toUpperCase()}</div>
                <div className="text-xs mt-0.5">{d.attempted ? `${d.score}%` : t(UI.noAttempt, lang)}</div>
              </div>
            ))}
          </div>
        </div>

        <div className="bg-white rounded-2xl p-6 bk-card">
          <div className="flex items-center gap-2 mb-4">
            <Clock size={16} className="bk-gold" />
            <div className="bk-display text-lg">{t(UI.readingHistory, lang)}</div>
          </div>
          <div className="flex flex-col divide-y bk-border">
            {bookHistory.map((h, i) => (
              <div key={i} className="flex items-center justify-between py-2.5 text-sm gap-2">
                <span className="bk-slate text-xs w-24 shrink-0">{h.date}</span>
                <span className="flex-1">{t(UI[MODE_LABELS[h.mode]], lang)}</span>
                <span className="bk-mono">{h.score}%</span>
              </div>
            ))}
          </div>
        </div>
      </div>
    </div>
  );
}

const WIZARD_STEPS = ["Create Book", "Upload Cover", "Short Summary", "Extended Summary", "Bloom Assessment", "Review", "Publish"];

        <div className="max-w-xl mx-auto">

            <ArrowLeft size={14} /> Cancel
          </button>

          <div className="flex flex-wrap gap-2 mb-6">
            {WIZARD_STEPS.map((s, i) => (
              <span key={s} className={`bk-mono text-[9px] px-2 py-1 rounded-full ${i === wizardStep ? "bk-bg-night bk-cream-text" : i < wizardStep ? "bk-bg-sage text-white" : "bg-[#EDE6D3] bk-slate"}`}>

                <label className="bk-mono text-[10px] bk-slate">AUTHOR</label>

          </div>
          {books.map((b) => (
            <div key={b.id} className="grid grid-cols-[1fr_auto_auto] gap-4 px-5 py-4 items-center border-b bk-border last:border-0">
              <div>
                <div className="text-sm font-medium">{b.title}</div>
                <div className="bk-slate text-xs">{b.author}</div>
              </div>
              <span className={`bk-mono text-[9px] px-2 py-1 rounded-full ${b.status === "available" ? "bk-bg-sage text-white" : "bg-[#EDE6D3] bk-slate"}`}>
                {b.status === "available" ? "PUBLISHED" : "COMING SOON"}
              </span>
              <span className="bk-mono text-[9px] bk-slate flex items-center gap-1">
                {b.status === "available" && <Lock size={11} />} {b.status === "available" ? "LOCKED" : "—"}
              </span>
            </div>
          ))}
        </div>
      </div>
    </div>
  );
}

export default function BookalyzerDemo() {
  const [role, setRole] = useState("select");
  const [resetKey, setResetKey] = useState(0);
  const [lang, setLang] = useState("en");

  return (
    <div className="bk-root w-full min-h-screen" key={resetKey}>
      <GlobalStyle />
      <TopNav role={role} setRole={setRole} onReset={() => { setResetKey((k) => k + 1); setRole("select"); }} lang={lang} setLang={setLang} />
      {role === "select" && <RoleSelect setRole={setRole} lang={lang} />}
      {role === "child" && <ChildApp key={`child-${resetKey}`} lang={lang} />}
      {role === "parent" && <ParentApp key={`parent-${resetKey}`} lang={lang} />}
      {role === "admin" && <AdminApp key={`admin-${resetKey}`} />}
    </div>
  );
}
